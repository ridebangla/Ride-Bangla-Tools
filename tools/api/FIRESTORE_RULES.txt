// Firestore rules for the self-hosted shortener.
// Firebase Console -> Firestore Database -> Rules -> paste inside `rules_version = '2';`
// then Publish. (Keep your existing rules; just ADD this block.)

match /rb_short_links/{code} {
  // Anyone can follow a short link (the redirect function reads the URL)
  allow read: if true;
  // Anyone can create, but the URL must be a sane http(s) string
  allow create: if request.resource.data.url is string
    && request.resource.data.url.size() > 0
    && request.resource.data.url.size() < 2000
    && request.resource.data.url.matches('^https?://.+');
  // Only the click counter may be updated (by the redirect function)
  allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['clicks']);
  allow delete: if false;
}
